# server.sh for docker and .env 
#!/bin/bash

JAR="/opt/apps/swordage-0.0.1-SNAPSHOT.jar"
ENV_FILE="/opt/apps/.env"

# Load environment variables from .env
if [ -f "$ENV_FILE" ]; then
    set -a
    source "$ENV_FILE"
    set +a
else
    echo "ERROR: .env file not found at $ENV_FILE"
    exit 1
fi

COMMAND="$1"

start_server() {
    echo "Starting..."
    nohup java -jar "$JAR" > app.log 2>&1 &
    echo $! > server.pid
    echo "Started with PID $(cat server.pid)"
}

stop_server() {
    if [ -f server.pid ]; then
        PID=$(cat server.pid)

        if kill -0 "$PID" 2>/dev/null; then
            echo "Stopping PID $PID..."
            kill "$PID"
            rm -f server.pid
            echo "Stopped."
        else
            echo "Process $PID is not running."
            rm -f server.pid
        fi
    else
        echo "Server not running."
    fi
}

restart_server() {
    stop_server
    sleep 2
    start_server
}

if [ "$COMMAND" == "start" ]; then
    start_server
elif [ "$COMMAND" == "stop" ]; then
    stop_server
elif [ "$COMMAND" == "restart" ]; then
    restart_server
else
    echo "Usage: ./server.sh start | stop | restart"
fi

```bash
#!/bin/bash

# ============================================================
# Zivdah Backend Server
# Docker Compose + .env
# ============================================================

set -e

APP_DIR="/opt/new_zivdah_backend"
COMPOSE_FILE="$APP_DIR/docker-compose.yml"
ENV_FILE="$APP_DIR/.env"

# ------------------------------------------------------------
# Go to application directory
# ------------------------------------------------------------
cd "$APP_DIR" || {
    echo "ERROR: Cannot access $APP_DIR"
    exit 1
}

# ------------------------------------------------------------
# Check required files
# ------------------------------------------------------------
if [ ! -f "$COMPOSE_FILE" ]; then
    echo "ERROR: docker-compose.yml not found at:"
    echo "$COMPOSE_FILE"
    exit 1
fi

if [ ! -f "$ENV_FILE" ]; then
    echo "ERROR: .env file not found at:"
    echo "$ENV_FILE"
    exit 1
fi

# ------------------------------------------------------------
# Docker check
# ------------------------------------------------------------
if ! command -v docker >/dev/null 2>&1; then
    echo "ERROR: Docker is not installed."
    exit 1
fi

if ! docker info >/dev/null 2>&1; then
    echo "ERROR: Docker is not running or current user has no Docker access."
    exit 1
fi

COMMAND="$1"

# ------------------------------------------------------------
# Functions
# ------------------------------------------------------------

start_server() {
    echo "=========================================="
    echo "Starting Zivdah Backend"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        up -d

    echo ""
    echo "Services:"
    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        ps
}

stop_server() {
    echo "=========================================="
    echo "Stopping Zivdah Backend"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        stop

    echo "Backend stopped."
}

restart_server() {
    echo "=========================================="
    echo "Restarting Zivdah Backend"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        restart

    echo ""
    echo "Services:"
    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        ps
}

update_server() {
    echo "=========================================="
    echo "Updating Zivdah Backend"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        up -d --build

    echo ""
    echo "Services:"
    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        ps
}

rebuild_server() {
    echo "=========================================="
    echo "Rebuilding Zivdah Backend"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        down

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        build --no-cache

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        up -d

    echo ""
    echo "Services:"
    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        ps
}

status_server() {
    echo "=========================================="
    echo "Zivdah Backend Status"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        ps
}

logs_server() {
    if [ -n "$2" ]; then
        docker compose \
            --env-file "$ENV_FILE" \
            -f "$COMPOSE_FILE" \
            logs --tail=100 "$2"
    else
        docker compose \
            --env-file "$ENV_FILE" \
            -f "$COMPOSE_FILE" \
            logs --tail=100
    fi
}

follow_logs() {
    if [ -n "$2" ]; then
        docker compose \
            --env-file "$ENV_FILE" \
            -f "$COMPOSE_FILE" \
            logs -f "$2"
    else
        docker compose \
            --env-file "$ENV_FILE" \
            -f "$COMPOSE_FILE" \
            logs -f
    fi
}

down_server() {
    echo "=========================================="
    echo "Stopping and Removing Zivdah Containers"
    echo "=========================================="

    docker compose \
        --env-file "$ENV_FILE" \
        -f "$COMPOSE_FILE" \
        down

    echo "Containers removed."
    echo "Docker volumes were NOT removed."
}

# ------------------------------------------------------------
# Command handling
# ------------------------------------------------------------

case "$COMMAND" in

    start)
        start_server
        ;;

    stop)
        stop_server
        ;;

    restart)
        restart_server
        ;;

    update)
        update_server
        ;;

    rebuild)
        rebuild_server
        ;;

    status)
        status_server
        ;;

    logs)
        logs_server "$@"
        ;;

    follow)
        follow_logs "$@"
        ;;

    down)
        down_server
        ;;

    *)
        echo ""
        echo "Zivdah Backend Server"
        echo ""
        echo "Usage:"
        echo "  ./server.sh start"
        echo "  ./server.sh stop"
        echo "  ./server.sh restart"
        echo "  ./server.sh update"
        echo "  ./server.sh rebuild"
        echo "  ./server.sh status"
        echo "  ./server.sh logs"
        echo "  ./server.sh logs user-service"
        echo "  ./server.sh follow"
        echo "  ./server.sh follow user-service"
        echo "  ./server.sh down"
        echo ""
        echo "Commands:"
        echo "  start    Start all containers"
        echo "  stop     Stop containers"
        echo "  restart  Restart containers"
        echo "  update   Build changed images and start"
        echo "  rebuild  Full no-cache rebuild"
        echo "  status   Show service status"
        echo "  logs     Show last 100 log lines"
        echo "  follow   Follow live logs"
        echo "  down     Stop and remove containers"
        echo ""
        exit 1
        ;;

esac
```
